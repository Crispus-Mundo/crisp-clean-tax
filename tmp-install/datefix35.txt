/** DATE FIX v3.5 (Nov 6, 2026) - ADDITIVE. Load after Growth.gs.
 * The spreadsheet turns "2026-11-13" into a date, so it reads back as "2026-11-13 00:00:00" (or a Date).
 * gParse_ split that on "-" and got NaN, so day counts came back empty: clients saw "today" on dates
 * months away, and the 7am Deadline Guard digest could miss items due in the next 7 days.
 * This reads the first yyyy-mm-dd in the value (or the Date itself) and keeps noon local time as before. */
var _df_parse = gParse_;
gParse_ = function (ymd) {
  if (Object.prototype.toString.call(ymd) === '[object Date]' && !isNaN(ymd)) return new Date(ymd.getFullYear(), ymd.getMonth(), ymd.getDate(), 12, 0, 0);
  var m = String(ymd || '').match(/(\d{4})-(\d{1,2})-(\d{1,2})/);
  if (!m) return _df_parse(ymd);
  return new Date(Number(m[1]), Number(m[2]) - 1, Number(m[3]), 12, 0, 0);
};

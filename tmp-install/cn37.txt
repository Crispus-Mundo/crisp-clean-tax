/******************************************************************************
 * CRISP CLEAN CLIENT PORTAL - ClientNotices.gs  (v3.5 Client notices pop-up, Nov 6, 2026)
 * ADDITIVE. New file only. Nothing in any other .gs file is edited or removed.
 * Like Suggestions.gs, this file wraps route_ and clientBundle_ and calls straight
 * through to the originals for everything it does not own. Load order: LAST file.
 * The only other change is one new <script> block added at the end of Index.html.
 *
 * What it does
 *  - A "notice" is a pop-up that opens when the client signs in. It shows an
 *    introduction, the documents they are confirming they received, the items
 *    they are agreeing to (one checkbox each), the terms (initials on each),
 *    a choice (for example the payment schedule), and a typed signature.
 *  - Notices are created HELD. Nothing reaches the client until Cyars releases it
 *    (run nRelease with the notice Id, or ask Claude to). That is her approval rule.
 *  - When the client signs: answers are saved in the Notices tab, a dated signed
 *    copy is saved to the client's Drive folder, a copy is emailed to the client,
 *    and Cyars is alerted. The client may instead send a question; Cyars is alerted.
 *  - Speed cache (PortalSpeed.gs) is respected: all writes use insert_/update_.
 *
 * Firm tools (run from the editor, no underscore so they show in the Run list):
 *   nList()            - logs every notice with its Id and status
 *   nRelease()         - releases the newest HELD notice (or set NOTICE_RELEASE_ID)
 *   nHold()            - puts a released notice back on hold (set NOTICE_RELEASE_ID)
 *   nSetupOmar()       - one-time: creates Omar Akbar Khan's scope change notice, HELD
 ******************************************************************************/
var NOTICE_V = 'v3.5-notices';
var NOTICE_RELEASE_ID = '';   // optional: put a notice Id here before running nRelease / nHold

var SHEETS_N = {
  Notices: ['Id', 'ClientId', 'Title', 'Intro', 'DocsJson', 'ItemsJson', 'TermsJson', 'ChoiceJson', 'Status', 'By',
    'CreatedAt', 'ReleasedAt', 'ShownAt', 'SignedAt', 'SignedName', 'SignedBy', 'Answers', 'Choice',
    'Question', 'QuestionAt', 'ReceiptFileId', 'EmailedTo']
};
Object.keys(SHEETS_N).forEach(function (k) { if (!SHEETS[k]) SHEETS[k] = SHEETS_N[k]; });

function nEnsure_() {
  var props = PropertiesService.getScriptProperties();
  if (props.getProperty('N_READY') === NOTICE_V) return;
  var lock = LockService.getScriptLock();
  if (!lock.tryLock(20000)) return;
  try {
    if (props.getProperty('N_READY') === NOTICE_V) return;
    var ss = ss_();
    Object.keys(SHEETS_N).forEach(function (n) { ensureSheet_(ss, n, SHEETS_N[n]); try { delete MEMO_['r:' + n]; } catch (e) {} });
    props.setProperty('N_READY', NOTICE_V);
    audit_('system', 'upgrade', NOTICE_V + ' Notices tab');
  } finally { lock.releaseLock(); }
}

function nJson_(s, d) { try { var v = JSON.parse(String(s || '')); return v == null ? d : v; } catch (e) { return d; } }

function nOut_(r) {
  return {
    id: r.Id, title: r.Title, intro: r.Intro, status: r.Status,
    docs: nJson_(r.DocsJson, []), items: nJson_(r.ItemsJson, []), terms: nJson_(r.TermsJson, []),
    choice: nJson_(r.ChoiceJson, null), signedAt: r.SignedAt || '', signedName: r.SignedName || '',
    chosen: r.Choice || '', question: r.Question || '', receiptFileId: r.ReceiptFileId || ''
  };
}

function nClientData_(clientId) {
  return byClient_('Notices', clientId).filter(function (r) {
    return r.Status === 'released' || r.Status === 'signed';
  }).map(nOut_);
}

function nPortalLink_() {
  return 'https://crispus-mundo.github.io/crisp-clean-tax/my-portal/';
}

/* ---------------- client actions ---------------- */
function nMine_(s, id) {
  var r = findRow_('Notices', id);
  if (!r || r.ClientId !== s.clientId || (r.Status !== 'released' && r.Status !== 'signed')) fail_('Notice not found.');
  return r;
}

function nSeen_(s, p) {
  var r = nMine_(s, p.id);
  if (!r.ShownAt) update_('Notices', r.Id, { ShownAt: now_() });
  return { list: nClientData_(s.clientId) };
}

function nAsk_(s, p) {
  var r = nMine_(s, p.id), txt = String(p.text || '').trim().slice(0, 3000);
  if (!txt) fail_('Type your question.');
  update_('Notices', r.Id, { Question: txt, QuestionAt: now_() });
  var c = findRow_('Clients', s.clientId), who = c ? c.DisplayName : s.clientId;
  try {
    notifyTeam_('Notice question: ' + who, who + ' has a question about "' + r.Title + '":\n\n' +
      (typeof scrubAll_ === 'function' ? scrubAll_(txt) : txt) + '\n\nOpen the portal (Firm) to answer.');
  } catch (e) {}
  try { audit_(s.person || 'client', 'notice-question', r.Id); } catch (e) {}
  return { list: nClientData_(s.clientId) };
}

function nSign_(s, p) {
  var r = nMine_(s, p.id);
  if (r.Status === 'signed') fail_('This notice is already signed. A copy is in your file.');
  var docs = nJson_(r.DocsJson, []), items = nJson_(r.ItemsJson, []), terms = nJson_(r.TermsJson, []), choice = nJson_(r.ChoiceJson, null);
  var gotDocs = [].concat(p.docs || []).map(String), gotItems = [].concat(p.items || []).map(String), ini = p.initials || {};
  docs.forEach(function (d) { if (gotDocs.indexOf(String(d.id)) < 0) fail_('Please confirm you received: ' + d.title); });
  items.forEach(function (it) { if (it.required !== false && gotItems.indexOf(String(it.id)) < 0) fail_('Please check: ' + it.label); });
  terms.forEach(function (t) {
    var v = String(ini[t.id] || '').trim();
    if (!/^[A-Za-z][A-Za-z.\- ]{0,5}$/.test(v)) fail_('Please initial each term.');
  });
  var chosenLabel = '';
  if (choice && choice.options && choice.options.length) {
    var pick = String(p.choice || '');
    choice.options.forEach(function (o) { if (String(o.id) === pick) chosenLabel = o.label; });
    if (!chosenLabel) fail_('Please choose: ' + choice.label);
  }
  var name = String(p.name || '').trim().slice(0, 120);
  if (name.length < 3) fail_('Type your full name to sign.');

  var c = findRow_('Clients', s.clientId), when = now_();
  var lines = [];
  lines.push('SIGNED NOTICE (informed consent)');
  lines.push('');
  lines.push('Client: ' + (c ? c.DisplayName : s.clientId));
  lines.push('Notice: ' + r.Title + ' (' + r.Id + ')');
  lines.push('Signed by: ' + name + ' (portal login: ' + (s.person || s.label || 'client') + ')');
  lines.push('Date and time: ' + when);
  lines.push('');
  lines.push('Documents I confirm I received:');
  docs.forEach(function (d) { lines.push('  [x] ' + d.title); });
  lines.push('');
  lines.push('Items I agreed to:');
  items.forEach(function (it) { lines.push('  [' + (gotItems.indexOf(String(it.id)) >= 0 ? 'x' : ' ') + '] ' + it.label + (it.price ? ' - ' + it.price : '')); });
  lines.push('');
  lines.push('Terms I initialed:');
  terms.forEach(function (t) { lines.push('  ' + String(ini[t.id]).trim().toUpperCase() + '  ' + t.text); });
  if (chosenLabel) { lines.push(''); lines.push((choice.label || 'Choice') + ': ' + chosenLabel); }
  lines.push('');
  lines.push('Introduction shown to me:');
  lines.push(String(r.Intro || ''));
  var text = lines.join('\n');

  var fileId = '';
  try {
    if (c && c.FolderId) {
      fileId = DriveApp.getFolderById(c.FolderId).createFile('Signed notice ' + today_() + ' - ' + String(r.Title).slice(0, 60) + '.txt',
        text + '\n', MimeType.PLAIN_TEXT).getId();
    }
  } catch (e) { try { audit_('system', 'notice-receipt-failed', e.message); } catch (x) {} }

  var sentTo = [];
  try {
    sentTo = clientEmails_(s.clientId);
    if (sentTo.length) {
      var body = 'Thank you. Here is your signed copy of "' + r.Title + '". It is also saved in your portal file.\n\n' + text +
        '\n\nOpen your portal: ' + nPortalLink_() +
        '\n\nIf anything here does not match what we discussed, tell us in writing so we can correct it in writing.' +
        '\n\nConfidential tax information protected under Internal Revenue Code section 7216.';
      MailApp.sendEmail({ to: sentTo.join(','), subject: 'Your signed copy: ' + r.Title, body: body + '\n\n' + adaFooter_(),
        htmlBody: emailHtml_(body), name: setting_('BRAND_NAME') || 'Crisp Clean' });
    }
  } catch (e) { try { audit_('system', 'notice-email-failed', e.message); } catch (x) {} }

  update_('Notices', r.Id, {
    Status: 'signed', SignedAt: when, SignedName: name, SignedBy: s.person || s.label || 'client',
    Answers: JSON.stringify({ docs: gotDocs, items: gotItems, initials: ini }), Choice: chosenLabel,
    ReceiptFileId: fileId, EmailedTo: sentTo.join(', ')
  });
  try {
    notifyTeam_('Notice signed: ' + (c ? c.DisplayName : s.clientId),
      (c ? c.DisplayName : s.clientId) + ' signed "' + r.Title + '".' + (chosenLabel ? '\nChoice: ' + chosenLabel : '') +
      '\nSigned copy saved to their Drive folder' + (sentTo.length ? ' and emailed to them.' : '.') + '\n\nOpen the portal (Firm) to see it.');
  } catch (e) {}
  try { audit_(s.person || 'client', 'notice-signed', r.Id + ' ' + r.Title); } catch (e) {}
  return { list: nClientData_(s.clientId), ok: true };
}

/* ---------------- wrappers ---------------- */
var N_CLIENT = { nSeen: 1, nSign: 1, nAsk: 1 };
if (typeof ACT_LABEL === 'object' && ACT_LABEL) { ACT_LABEL.nSeen = 'Opened a notice'; ACT_LABEL.nSign = 'Signed a notice'; ACT_LABEL.nAsk = 'Asked about a notice'; }

var _n_route = route_;
route_ = function (action, p, token) {
  if (!N_CLIENT[action]) return _n_route(action, p, token);
  nEnsure_();
  var s = session_(token);
  if (!s) fail_('Your session ended. Please log in again.');
  if (s.role !== 'client') fail_('That action is not available.');
  p = p || {};
  var out = action === 'nSeen' ? nSeen_(s, p) : action === 'nAsk' ? nAsk_(s, p) : nSign_(s, p);
  try { logActivity_(s, action, {}); } catch (e) {}
  return out;
};

var _n_bundle = clientBundle_;
clientBundle_ = function (clientId, s) {
  var b = _n_bundle(clientId, s);
  try { nEnsure_(); b.notices = nClientData_(clientId); } catch (e) { b.notices = []; }
  return b;
};

/* ---------------- firm tools (run from the editor) ---------------- */
function nList() {
  nEnsure_();
  var out = rows_('Notices').map(function (r) { return [r.Id, r.ClientId, r.Title, r.Status, r.ReleasedAt || '', r.SignedAt || ''].join(' | '); });
  Logger.log(out.join('\n') || 'No notices yet.');
  return out.join('\n');
}

function nRelease() {
  nEnsure_();
  var id = NOTICE_RELEASE_ID, list = rows_('Notices');
  var r = id ? findRow_('Notices', id) : list.filter(function (x) { return x.Status === 'held'; }).pop();
  if (!r) { Logger.log('No held notice found.'); return 'No held notice found.'; }
  if (r.Status !== 'held') { Logger.log('Notice ' + r.Id + ' is ' + r.Status + ', not held.'); return r.Status; }
  update_('Notices', r.Id, { Status: 'released', ReleasedAt: now_() });
  audit_('Cyars', 'notice-released', r.Id + ' ' + r.Title);
  Logger.log('Released: ' + r.Id + ' ' + r.Title);
  return 'Released ' + r.Id;
}

function nHold() {
  nEnsure_();
  var r = NOTICE_RELEASE_ID ? findRow_('Notices', NOTICE_RELEASE_ID) : null;
  if (!r) { Logger.log('Set NOTICE_RELEASE_ID first.'); return 'Set NOTICE_RELEASE_ID first.'; }
  if (r.Status === 'signed') { Logger.log('Already signed; left as is.'); return 'signed'; }
  update_('Notices', r.Id, { Status: 'held' });
  audit_('Cyars', 'notice-held', r.Id);
  Logger.log('On hold: ' + r.Id);
  return 'held';
}

/******************************************************************************
 * CRISP CLEAN CLIENT PORTAL - EstateReviews.gs  (v3.9 Estate readiness + Reviews, Nov 7, 2026)
 * ADDITIVE. New file only. Wraps route_, clientBundle_ and doGet and calls straight through to the
 * originals for everything it does not own. Load after CaseUpdates.gs / TeamAppLinksLate.gs.
 *
 * 1. ESTATE READINESS (client): a checklist of what the household has (will, trust, financial POA,
 *    Michigan patient advocate designation, funeral representative, guardian for children,
 *    beneficiaries, transfer-on-death, home title, digital accounts, letter of instruction,
 *    business succession), a list of assets and who they go to (no account numbers), and a note for
 *    their attorney. EDUCATION AND ORGANIZING ONLY. The portal never drafts or reviews wills,
 *    trusts or powers of attorney: in Michigan that is the practice of law (State Bar of Michigan).
 * 2. ATTORNEY HANDOFF: the client chooses a Michigan attorney (from the firm's list or their own),
 *    picks what to share and signs a one-time consent (IRC 7216 / Rev. Proc. 2013-14 wording).
 *    The request waits for Cyars. Nothing goes out until she taps Send. Send = a branded email
 *    to the attorney with an "Open the readiness packet" button (secure link, expires in 30 days,
 *    every open logged, can be revoked) + a PDF copy in the client's Drive folder + client log.
 * 3. REVIEW REQUESTS: owner sets the Google review link; asks clients (one, or everyone not yet
 *    asked) with the SAME message for everyone (Google's rules bar asking only happy clients;
 *    FTC 16 CFR Part 465 bars buying or faking reviews). The button goes through a tracking page
 *    (?rv=) so she sees who opened it; the client can also tap "I posted it" or "Not now" in the
 *    portal. Tracker shows asked / opened / posted / not now.
 * Owner login only for firm actions. All writes use insert_/update_ (speed cache safe).
 * Link words used: pk (packet) and rv (review). Never "sid" or "c" (reserved by Apps Script).
 ******************************************************************************/

function erEnsure_() {
  var props = PropertiesService.getScriptProperties();
  if (props.getProperty('ER_READY') === ER_V) return;
  var lock = LockService.getScriptLock();
  if (!lock.tryLock(20000)) return;
  try {
    if (props.getProperty('ER_READY') === ER_V) return;
    var ss = ss_();
    Object.keys(SHEETS_ER).forEach(function (n) { ensureSheet_(ss, n, SHEETS_ER[n]); delete MEMO_['r:' + n]; });
    props.setProperty('ER_READY', ER_V);
    audit_('system', 'upgrade', ER_V + ' EstatePlans, EstateHandoffs, ReviewRequests tabs');
  } finally { lock.releaseLock(); }
}
function erTxt_(v, n) { return String(v == null ? '' : v).replace(/\r/g, '').slice(0, n || 2000); }
/** Long digit runs (account, policy or Social Security numbers) never get stored. */
function erNoNumbers_(s) { return String(s || '').replace(/\d[\d\s\-]{5,}\d/g, function (m) { return m.replace(/\d/g, '•'); }); }

/* ---------------- estate plan (client) ---------------- */
function erPlanRow_(clientId) { return findRow_('EstatePlans', 'ep_' + clientId); }
function erPlan_(clientId) {
  var r = erPlanRow_(clientId);
  return { answers: r ? erJson_(r.Answers, {}) : {}, assets: r ? erJson_(r.Assets, []) : [], notes: r ? (r.Notes || '') : '', updatedAt: r ? r.UpdatedAt : '', updatedBy: r ? r.UpdatedBy : '' };

function erScore_(answers) {
  var app = 0, have = 0, open = 0;
  ER_ITEMS.forEach(function (it) { var a = answers[it[0]] || {}; if (a.s === 'na') return; app++; if (a.s === 'have') have++; else open++; });
  return { applicable: app, have: have, open: open, pct: app ? Math.round(have * 100 / app) : 0, answered: ER_ITEMS.filter(function (it) { return (answers[it[0]] || {}).s; }).length };
}
function erSavePlan_(s, p) {
  var answers = {}, inA = p.answers || {};
  ER_ITEMS.forEach(function (it) {
    var a = inA[it[0]] || {}; if (!ER_STATES[a.s]) return;
    answers[it[0]] = { s: a.s, year: erTxt_(a.year, 4).replace(/[^0-9]/g, ''), where: erNoNumbers_(erTxt_(a.where, 160)), who: erNoNumbers_(erTxt_(a.who, 120)) };
  });
  var assets = (p.assets || []).slice(0, 60).map(function (x) {
    return { type: erTxt_(x.type, 40), desc: erNoNumbers_(erTxt_(x.desc, 160)), value: erTxt_(x.value, 40), title: erTxt_(x.title, 60), bene: erNoNumbers_(erTxt_(x.bene, 160)) };
  }).filter(function (x) { return x.desc || x.bene; });
  var who = s.person || s.label || 'Client', fields = { Answers: JSON.stringify(answers), Assets: JSON.stringify(assets), Notes: erNoNumbers_(erTxt_(p.notes, 3000)), UpdatedAt: now_(), UpdatedBy: who };
  var r = erPlanRow_(s.clientId);
  if (r) update_('EstatePlans', r.Id, fields);
  else { fields.Id = 'ep_' + s.clientId; fields.ClientId = String(s.clientId); insert_('EstatePlans', fields); }
  var sc = erScore_(answers);
  try { clientLog_(s.clientId, { grp: 'Estate readiness', source: 'client portal', actor: who, action: 'Updated the estate readiness checklist', detail: sc.have + ' of ' + sc.applicable + ' in place, ' + assets.length + ' assets listed' }); } catch (e) {}
  return erClientData_(s.clientId, s);
}

/* ---------------- attorney handoff (client) ---------------- */
function erAttorneys_() {
  return String(setting_('ESTATE_ATTORNEYS') || '').split('\n').map(function (l) {
    var x = l.split('|').map(function (y) { return y.trim(); }); if (!x[0] || !x[2]) return null;
    return { name: x[0], firm: x[1] || '', email: x[2], phone: x[3] || '', id: hash_('att|' + x[2].toLowerCase()).slice(0, 10) };
  }).filter(Boolean);
}

function erRequest_(s, p) {
  var b = brand_(), att = null;
  if (p.attId) att = erAttorneys_().filter(function (a) { return a.id === p.attId; })[0];
  if (!att) {
    att = { name: erTxt_(p.name, 120).trim(), firm: erTxt_(p.firm, 120).trim(), email: erTxt_(p.email, 160).trim().toLowerCase(), phone: erTxt_(p.phone, 30).trim() };
    if (!att.name) fail_('Type your attorney\'s name.');
    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(att.email)) fail_('Type your attorney\'s email address.');
  }
  var share = ['checklist', 'assets', 'note'].filter(function (k) { return p['sh_' + k]; });
  if (!share.length) fail_('Pick at least one thing to share.');
  var signed = erTxt_(p.signed, 120).trim();
  if (signed.length < 4 || signed.split(/\s+/).length < 2) fail_('Type your full name to sign.');
  if (!p.agree) fail_('Check the box to agree.');
  if (rows_('EstateHandoffs').some(function (h) { return h.ClientId === String(s.clientId) && h.Status === 'requested'; })) fail_('You already have a request waiting. Withdraw it first if you want to change it.');
  var text = erConsentText_(signed, att, share, b), at = now_();
  var row = insert_('EstateHandoffs', { Id: newId_('eh'), ClientId: String(s.clientId), Person: s.person || '', AttorneyName: att.name, AttorneyFirm: att.firm, AttorneyEmail: att.email,
    AttorneyPhone: att.phone, Share: share.join(','), ClientNote: '', SignedName: signed, ConsentText: text, ConsentAt: at, Status: 'requested', DecidedAt: '', DecidedBy: '', DeclineNote: '',
    LinkKey: '', LinkExpires: '', Opens: 0, LastOpenAt: '', EmailedTo: '', DriveFileId: '', UpdatedAt: at });
  var c = findRow_('Clients', s.clientId), who = s.person || s.label || 'Client';
  try { clientLog_(s.clientId, { grp: 'Estate readiness', source: 'client portal', actor: who, action: 'Signed consent to send the readiness packet to ' + att.name, detail: text }); } catch (e) {}
  try { notifyTeam_('Attorney handoff to approve: ' + (c ? c.DisplayName : s.clientId), (c ? c.DisplayName : s.clientId) + ' asked us to send their estate readiness packet to ' + att.name + (att.firm ? ' (' + att.firm + ')' : '') +
    '. Nothing has gone out. Review and tap Send in the portal (Firm login > Estate handoffs).\n\nOpen the portal: ' + erUrl_()); } catch (e) {}
  audit_(who, 'estate-handoff-requested', s.clientId + ' ' + row.Id);
  return erClientData_(s.clientId, s);
}
function erWithdraw_(s, p) {
  var h = findRow_('EstateHandoffs', p.id);
  if (!h || h.ClientId !== String(s.clientId)) fail_('That request is not available.');
  if (h.Status !== 'requested') fail_('That request was already ' + h.Status + '.');
  update_('EstateHandoffs', h.Id, { Status: 'withdrawn', DecidedAt: now_(), DecidedBy: s.person || s.label || 'Client', UpdatedAt: now_() });
  try { clientLog_(s.clientId, { grp: 'Estate readiness', source: 'client portal', actor: s.person || s.label || 'Client', action: 'Withdrew the attorney handoff request', detail: h.AttorneyName }); } catch (e) {}
  return erClientData_(s.clientId, s);
}

/* ---------------- review requests (client) ---------------- */
function erReviewFor_(clientId) {
  var list = rows_('ReviewRequests').filter(function (r) { return r.ClientId === String(clientId); });
  return list.length ? list[list.length - 1] : null;
}
function erReviewAnswer_(s, p) {
  var r = erReviewFor_(s.clientId); if (!r) fail_('Nothing to answer here.');
  var a = p.answer === 'posted' ? 'posted' : p.answer === 'later' ? 'later' : 'opened';
  var f = { UpdatedAt: now_(), Answer: a };
  if (a === 'posted') { f.Status = 'posted'; f.PostedAt = now_(); }
  else if (a === 'later') { if (r.Status !== 'posted') f.Status = 'not now'; }
  else { if (r.Status === 'asked') f.Status = 'opened'; if (!r.ClickedAt) f.ClickedAt = now_(); }
  update_('ReviewRequests', r.Id, f);
  var link = a === 'opened' ? setting_('REVIEW_LINK') : '';
  var out = erClientData_(s.clientId, s); out.openLink = link; return out;
}

function erClientData_(clientId, s) {
  var plan = erPlan_(clientId), rv = erReviewFor_(clientId);
  return {
    items: ER_ITEMS, states: ER_STATES, share: ER_SHARE, days: ER_DAYS, plan: plan, score: erScore_(plan.answers),
    attorneys: erAttorneys_().map(function (a) { return { id: a.id, name: a.name, firm: a.firm, phone: a.phone }; }),
    handoffs: rows_('EstateHandoffs').filter(function (h) { return h.ClientId === String(clientId) && h.Status !== 'deleted'; }).map(function (h) {
      return { id: h.Id, attorney: h.AttorneyName, firm: h.AttorneyFirm, status: h.Status, consentAt: h.ConsentAt, decidedAt: h.DecidedAt, expires: h.LinkExpires, note: h.Status === 'declined' ? (h.DeclineNote || '') : '', share: String(h.Share || '').split(',') };
    }).reverse(),
    review: rv && setting_('REVIEW_LINK') ? { status: rv.Status, askedAt: rv.AskedAt } : null
  };
}

function erSettings_(p) {
  if (p.attorneys != null) {
    var lines = String(p.attorneys).split('\n').map(function (l) { return l.trim(); }).filter(String).slice(0, 40);
    lines.forEach(function (l) { var x = l.split('|').map(function (y) { return y.trim(); }); if (!x[0] || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(x[2] || '')) fail_('Each line needs: Name | Firm | email | phone. Check: ' + l.slice(0, 60)); });
    putSetting_('ESTATE_ATTORNEYS', lines.join('\n'));
  }
  if (p.reviewLink != null) {
    var u = String(p.reviewLink).trim();
    if (u && !/^https:\/\/(g\.page|www\.google\.com|search\.google\.com|maps\.app\.goo\.gl|goo\.gl|g\.co)\//.test(u)) fail_('Paste the review link from your Google Business Profile (it starts with https://g.page/ or https://search.google.com/).');
    putSetting_('REVIEW_LINK', u);
  }
  return erAdminData_({ note: 'Saved.' });
}

  ids.forEach(function (id) {

    var to = erEmails_(id, ''); if (!to.length) { skipped.push(c.DisplayName + ' (no email)'); return; }
    if (MailApp.getRemainingDailyQuota() <= 5 + to.length) { skipped.push(c.DisplayName + ' (email limit)'); return; }
    var key = erKey_(), url = erUrl_() + '?rv=' + key, names = people_(c), first = firstName_(names[0] || c.DisplayName), sent = 0;
    to.forEach(function (addr) { try { if (erSend_(addr, 'Would you share your experience? — ' + b.name, erReviewText_(first), url, 'Leave a Google review')) sent++; } catch (e) { audit_('system', 'review-email-failed', e.message); } });

var _er_route = route_;
route_ = function (action, p, token) {
  if (!ER_CLIENT[action] && !ER_ADMIN[action]) return _er_route(action, p, token);
  erEnsure_();
  var s = session_(token);
  if (!s) fail_('Your session ended. Please log in again.');
  p = p || {};
  if (ER_CLIENT[action]) {
    if (s.role !== 'client') fail_('That action is not available.');
    var out = action === 'erSavePlan' ? erSavePlan_(s, p) : action === 'erRequest' ? erRequest_(s, p) : action === 'erWithdraw' ? erWithdraw_(s, p) : erReviewAnswer_(s, p);
    try { logActivity_(s, action, {}); } catch (e) {}
    return out;
  }
  if (s.role !== 'admin') fail_('Only the owner login can use this page.');
  if (typeof HARDEN_ === 'object') HARDEN_.firmApi = true;
  try {
    switch (action) {
      case 'erAdmin': return erAdminData_();
      case 'erPreview': return erPacketPreview_(p);
      case 'erApprove': return erApprove_(p, 'Cyars');
      case 'erDecline': return erDecline_(p, 'Cyars');
      case 'erRevoke': return erRevoke_(p, 'Cyars');
      case 'erSettings': return erSettings_(p);
      case 'erAsk': return erAsk_(p, 'Cyars');
      case 'erMark': return erMark_(p, 'Cyars');
    }
  } finally { if (typeof HARDEN_ === 'object') HARDEN_.firmApi = false; }
};

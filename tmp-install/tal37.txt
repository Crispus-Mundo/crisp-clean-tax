/** TeamAppLinksLate.gs (Nov 6, 2026) - ADDITIVE. The Team App Hub code from Nov 6 (saved in ScopeChanges.gs)
 * runs before TeamAccess.gs defines TEAM_APPS, so the hub card and app links never applied. This repeats the
 * same steps after TEAM_APPS exists, using the same TAL_LINKS_ list. Changes nothing else. Must load after TeamAccess.gs. */
(function () {
  try {
    if (typeof TEAM_APPS === 'undefined' || typeof TAL_LINKS_ === 'undefined') return;
    if (!TEAM_APPS.some(function (a) { return a.key === 'hub'; })) {
      TEAM_APPS.unshift({
        key: 'hub', title: 'Team App Hub (start here)',
        who: 'Your starting page for every app, sign-in and how-to video',
        why: 'One page with every app link, short how-to videos (with sound, full-screen), payroll steps for Homebase and Gusto, and your company email details.',
        does: 'Lists the clients assigned to you, lets you tick off a learning checklist, and opens each app in a new tab.',
        can: 'Watch the videos, open any of your apps, copy sign-in steps, and track your checklist.',
        held: 'Nothing here reaches a client.',
        signin: 'Open the link, then the same email and PIN you use here.',
        url: '__HUB_URL__'
      });
    }
    TEAM_APPS.forEach(function (a) { if (TAL_LINKS_[a.key] && String(a.url || '').indexOf('__') === 0) a.url = TAL_LINKS_[a.key]; });
  } catch (e) {}
})();
